const actualFetch = window.fetch.bind(window);
window.fetch = async (...arguments_) => {
  const response = await actualFetch(...arguments_);
  const input = arguments_[0];
  const url = new URL(input instanceof Request ? input.url : String(input), location.href);
  if (url.pathname === '/api/chat/retry') {
    const readBody = response.json.bind(response);
    response.json = async () => {
      const body = await readBody();
      evidence.retryBodyConsumed = (evidence.retryBodyConsumed ?? 0) + 1;
      return body;
    };
  }
  return response;
};

function ReceiptOwner() {
  const inventory = useInventoryCommand({
    worldId: 'component-world',
    actorId: 'player',
    timeline: 'timeline',
    scope: 'component-scope',
    recoveryScope: 'recovery',
    epoch: 'epoch',
    connected: true,
    command: async () => {
      throw Error('Native dispatch forbidden');
    },
    onResolved: () => {
      evidence.receiptsResolved++;
    },
  });
  return (
    <section aria-label="Actual inventory receipt owner">
      <p>{inventory.pending ? inventory.pending.status : 'resolved'}</p>
      <button onClick={() => void inventory.check()}>Check original receipt</button>
      <p>{inventory.message}</p>
    </section>
  );
}
function PersonFirst() {
  const trade = useTradeDraft(tradeKey, [], { key: inventoryKey, scope: tradeScope });
  const [recovery, setRecovery] = useState(false);
  const [outcome, setOutcome] = useState('not tried');
  return (
    <section aria-label="Actual person trade hook">
      <p>Original read blocked: {String(trade.inventoryReadBlocked)}</p>
      <p>Original receipt: {trade.inventoryReceipt ?? 'none'}</p>
      <button
        onClick={() => {
          const accepted = trade.start();
          evidence.tradeStarts.push(accepted);
          setOutcome(String(accepted));
        }}
      >
        Try person trade
      </button>
      <output aria-label="Person trade start result">{outcome}</output>
      <button onClick={() => setRecovery(true)}>Mount original receipt owner</button>
      {recovery && <ReceiptOwner />}
    </section>
  );
}
function Fixture() {
  const scenario = new URLSearchParams(location.search).get('scenario');
  const [npc, setNpc] = useState('A');
  const [revision, setRevision] = useState(1);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');
  const [closed, setClosed] = useState(false);
  const [draft, setDraft] = useState('Unsaved original draft');
  const [pageKey, setPageKey] = useState('original');
  if (scenario === 'trade') return <PersonFirst />;
  if (scenario === 'chat')
    return (
      <>
        <nav aria-label="Fixture parent conversation controls">
          <button
            onClick={() => {
              setNpc('B');
              setRevision((value) => value + 1);
            }}
          >
            Fixture: select Bea
          </button>
        </nav>
        <section aria-label="Actual conversation" style={{ width: 720, height: 700 }}>
          <Composer
            view={view}
            connected
            npcId={npc}
            entry={null}
            talkRevision={revision}
            chooseRecipient={setNpc}
            clearEntry={() => {}}
            setup={() => {
              throw Error('Provider setup is forbidden');
            }}
            notify={(text) => evidence.notices.push(text)}
            visible
          />
        </section>
      </>
    );
  if (scenario === 'editor')
    return closed ? (
      <p>Fixture editor closed</p>
    ) : (
      <EditorPanel
        title="Edit fixture"
        dirty
        saving={saving}
        error={error}
        scope="component-editor"
        tabs={[
          {
            id: 'identity',
            label: 'Identity',
            icon: 'ui.character',
            content: (
              <label>
                Fixture draft
                <input value={draft} onChange={(event) => setDraft(event.target.value)} />
              </label>
            ),
          },
        ]}
        onClose={() => {
          evidence.closes++;
          setClosed(true);
        }}
        onSave={() => {
          evidence.saveCalls++;
          setSaving(true);
          return new Promise((resolve) => {
            evidence.releaseSave = () => {
              setSaving(false);
              setError('Controlled save refused; draft retained.');
              resolve(false);
            };
          });
        }}
        onDiscard={() => {
          throw Error('Discard is not part of this observation');
        }}
      />
    );
  return (
    <>
      <button onClick={() => setPageKey('C-removed')}>Fixture: publish C removed</button>
      <section aria-label="Actual matching lots">
        <MergeTargets
          item={{
            id: 'A',
            name: 'Wood A',
            definitionId: 'wood',
            quantity: 1,
            revision: 1,
            placementRevision: 1,
            category: 'material',
            description: 'Source',
            equipped: false,
            tags: [],
            actions: [],
          }}
          containerId="player"
          pageKey={pageKey}
          canAct
          visible
          onMerge={(target) => evidence.merges.push(target.id)}
        />
      </section>
    </>
  );
}
createRoot(document.getElementById('app')!).render(<Fixture />);
